function findEvenIndex(arr) {
  const totalSum = arr.reduce((acc, el) => el + acc, 0);
  const sum = 0;
  for (let i = 0; i < arr.length; i++) {
    const firstPart = arr.slice(0, i);
    const sum1 = firstPart.reduce((acc, el) => el + acc, 0);
    const sum2 = totalSum - arr[i] - sum1;
    if (sum1 === sum2) {
      return i;
    }
  }
    return -1;
}
